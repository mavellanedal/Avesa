public async insertLead(createLead: CreateLeadDto): Promise<CreateLeadResponseDto> {
  try {
    const hasMultiSourceRole = UserUtil.getRoles().includes(ROLES.EXTERNAL_API_MULTI_SOURCE);
    const sourceId: string[] = UserUtil.getSourceIds();

    const needProvider = hasMultiSourceRole || sourceId.length > 1;
    if (needProvider && createLead.provider == null) {
      throw new ExternalApiException(ErrorMessages.isNotEmpty.replace("$property","provider"));
    }

    const platformSource: PlatformSource = new PlatformSource();
    if (createLead.gclid) {
      platformSource.platformId = LEAD_SYSTEM.PLATFORM.GOOGLE;
    } else if (createLead.fbclid ||createLead.formId) {
      platformSource.platformId = LEAD_SYSTEM.PLATFORM.META;
    } else if (createLead.ttclid) {
      platformSource.platformId = LEAD_SYSTEM.PLATFORM.TIKTOK;
    }

    let source: Source = null;
    if (needProvider) {
      source = await this.sourceService.sourceRepository.getSourceActiveByName(createLead.provider);
    } else {
      source = await this.sourceService.sourceRepository.getSourceActiveById(Number(sourceId[0]))
    }

    if (source) {
      platformSource.source = source;
    } else {
      this.logger.warn(JSON.stringify(createLead));
      throw new ExternalApiException(ExternalApiErrors.SOURCE_NOT_FOUND);
    }

    const newLead = this.mapper.map(createLead, CreateLeadDto, Lead);

    const lead = await this.leadService.createLead(newLead, platformSource);

    if (!lead) {
      throw new ExternalApiException(ExternalApiErrors.INSERT_LEAD);
    }

    const quality = await this.dispatcherService.checkQuality(lead);

    if (quality.error && this.qualityChecks[quality.error]) {
      const mapped = this.qualityChecks[quality.error](newLead);
      this.logger.warn(mapped.message);
      throw new ExternalApiException(mapped.error, mapped['status']);
    }

    if (quality.continueDispatch) {
      await this.leadService.createLeadStateHistory(lead, new Date(), LEAD_SYSTEM.LEAD_STATE.INTERESTED, LEAD_SYSTEM.LEAD_STATE.SUB_STATES.MQL);
      await this.dispatcherService.dispatchLead(lead);
    }

    return plainToInstance(CreateLeadResponseDto, lead, {strategy: 'excludeAll'});
  } catch (error) {
    this.logger.error(error);
    throw (error instanceof ExternalApiException) ? error : new ExternalApiException(ExternalApiErrors.INSERT_LEAD)
  }
}

  const rawSourceIds: string[] = UserUtil.getSourceIds();
  let sourceQueryLevelId: number | null = null

  if (rawSourceIds && rawSourceIds.length > 0) {
  const sourceIds = rawSourceIds.map(id => Number(id));
  sourceQueryLevelId = await this.sourceService.getSourceQueryLevel(sourceIds);
  if (sourceQueryLevelId === LEAD_SYSTEM.SOURCE.SOURCE_QUERY_LEVEL.SIN_CONSULTA) {
  throw new ExternalApiException(ExternalApiErrors.NOT_ACCESS_RESOURCE , HttpStatus.UNAUTHORIZED);
  }
  if (sourceQueryLevelId !== null) {
  const leadCode = leadInfoDto.leadCode;
  const leadSourceId = await this.leadService.getSourceIdByLeadCode(leadCode);

  if (!leadSourceId) {
  throw new ExternalApiException(ExternalApiErrors.LEAD_NOT_FOUND, HttpStatus.NOT_FOUND);
  }

  if (!sourceIds.includes(leadSourceId)) {
  throw new ExternalApiException(ExternalApiErrors.NOT_ACCESS_RESOURCE, HttpStatus.UNAUTHORIZED);
  }
  }
  }

  const leadDto = await this.leadService.getLeadInfo(leadInfoDto, sourceQueryLevelId);

  if (!leadDto) {
  return null;
  }
  leadDto.id = undefined;
  if (leadDto.leadSourceInfo) {
  leadDto.leadSourceInfo.leadId = undefined;
  }
  if (leadDto.leadStateHistories) {
  leadDto.leadStateHistories.forEach(h => h.id = undefined);
  }

  return leadDto;
}

import {Injectable, Logger, Inject} from "@nestjs/common";
import workflowConfig from "@modules/workflows/config/workflow.config";
import {ConfigType} from "@nestjs/config";
import {firstValueFrom} from "rxjs";
import {HttpService} from "@nestjs/axios";

@Injectable()
export class WorkflowService {
    private readonly logger = new Logger(WorkflowService.name);

    constructor(
        @Inject(workflowConfig.KEY) private configWorkflow: ConfigType<typeof workflowConfig>,
        private readonly httpService: HttpService
    ) {}

    public async sendWebhookCM(payload: Record<string, any>): Promise<void> {
        try {
            await firstValueFrom(
                this.httpService.post(this.configWorkflow.url, payload, {
                    headers: {
                        'Content-Type': 'application/json',
                        'x-api-key': this.configWorkflow.authToken
                    },
                })
            );
            this.logger.log(`Webhook enviado a n8n con éxito. Tipo: ${payload.type}, Acción: ${payload.action}.`)
        } catch (error) {
            const errorMsg = error.response?.data || error.message;
            this.logger.error(`Error al comunicarse con n8n (Payload: ${payload.type}) - Detalle: ${JSON.stringify(errorMsg)}.`);
        }
    }
}

        const newDestination = plainToInstance(Destination, destination);

        await this.createDestinationHistory(newDestination);

        action: LEAD_SYSTEM.WF.CM.ACTIONS.CREATE,

            default: dp.default
        })) || [],
